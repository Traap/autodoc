% Standard Design Review Abstract.
% ------------------------------------------------------------------------------
\newcommand{\stdDesignReview}{Design Review}
\newcommand{\stdDesignReviewTitle}[1]{#1}

\newcommand{\stdDesignReviewAbstract}[1]{%
  This document records the \stdDesignReview\ for #1.  It summarizes the design
  information reviewed, the decisions made, and any actions required to maintain
  compliance with applicable software development, risk management, and quality
  controls.  After approval, the record is retained in the \Gls{company}
  \stdQMS\ as objective evidence for design history and regulatory review.
}%

% ------------------------------------------------------------------------------
% Standard Design Review document generator command.
% ------------------------------------------------------------------------------
\newcommand{\stdDesignReviewDoc}[1]{%
  \stdAutoDocLayout{\stdDesignReviewTitle{#1}}%
                   {\stdDesignReviewAbstract{#1}}%
                   {\designReviewDir/main.tex}%
}%
