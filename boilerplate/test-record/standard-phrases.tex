% ------------------------------------------------------------------------------
% Standard Test Record Abstract.
% ------------------------------------------------------------------------------
\newcommand{\stdTestRecord}{Test Record}
\newcommand{\stdTestRecordTitle}[1]{#1}

\newcommand{\stdTestRecordAbstract}[1]{%
  #1 is a controlled \stdTestRecord\ that supports the \Gls{dhf}.  It identifies
  the \stdCfgItem\ under test, the applicable \stdRptPkg, the executed test
  reports, and the objective evidence used to determine whether the referenced
  requirements were satisfied.

  The record is verified when the \stdRptPkg\ has been executed, required test
  evidence has been obtained, deviations have been dispositioned, and the
  evidence supports the conclusion that the applicable \stdSyRS\ are satisfied.
  After verification, #1 is retained in the \Gls{company} \Gls{dms}.
}%

% ------------------------------------------------------------------------------
% Standard Test Record document generator command.
% ------------------------------------------------------------------------------
\newcommand{\stdTestRecordDoc}[1]{%
  \stdAutoDocLayout{\stdTestRecordTitle{#1}}%
                   {\stdTestRecordAbstract{#1}}%
                   {\testRecordDir/main.tex}%
}%
