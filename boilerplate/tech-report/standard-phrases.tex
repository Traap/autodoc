% ------------------------------------------------------------------------------
% Standard Technical Report Abstract.
% ------------------------------------------------------------------------------
\newcommand{\stdTechReport}{Technical Report}
\newcommand{\stdTechReportTitle}[1]{#1 \stdTechReport}

\newcommand{\stdTechReportAbstract}[1]{%
  This #1 \stdTechReport\ documents a controlled technical evaluation for the
  identified software or quality system topic.  It summarizes the objective,
  method, evidence reviewed, limitations, and conclusion so the result can be
  assessed by software, quality, and regulatory reviewers.  The completed report
  should be retained with the applicable \Gls{company} quality or design history
  records.
}%

% ------------------------------------------------------------------------------
% Standard Technical Report document generator command.
% ------------------------------------------------------------------------------
\newcommand{\stdTechReportDoc}[1]{%
  \stdAutoDocLayout{\stdTechReportTitle{#1}}%
                   {\stdTechReportAbstract{#1}}%
                   {\techReportDir/main.tex}%
}%
