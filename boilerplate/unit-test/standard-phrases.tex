% ------------------------------------------------------------------------------
% Standard Unit Test Abstract.
% ------------------------------------------------------------------------------
\newcommand{\stdUnitTypeTest}{Unit Test}
\newcommand{\stdUnitTest}[1]{#1 \stdUnitTypeTest}

\newcommand{\stdUnitTestAbstract}[1]{%
  This report documents unit testing for #1 as a controlled software lifecycle
  record.  It identifies the unit or component under test, the executed test
  evidence, deviations, and review conclusion used to determine whether the
  implementation satisfies the applicable low-level requirements or intended use
  expectations.  After verification, #1 is retained in the \Gls{company}
  \Gls{dms} as part of the supporting \Gls{dhf}.
}

% ------------------------------------------------------------------------------
% Standard Unit Test document generator command.
% ------------------------------------------------------------------------------
\newcommand{\stdUnitTestDoc}[1]{%
  \stdAutoDocLayout{\stdUnitTest{#1}}%
                   {\stdUnitTestAbstract{#1}}%
                   {\unitTestDir/main.tex}%
}%
