\newcommand{\stdVerTest}{Verification Test}
\newcommand{\stdVerTestTitle}[1]{#1 \stdVerTest}

\newcommand{\stdVerTestAbstract}[1]{%
  This document records verification of #1 as the \stdCfgItem\ identified in
  this \stdRptPkg.  It describes the applicable requirements, executed test
  reports, objective evidence, deviations, and review conclusion used to
  determine whether the \stdCfgItem\ satisfies the referenced \stdSyRS\ and
  \stdSRS.  After verification, #1 is retained in the \Gls{company} \stdQMS.
}%

% ------------------------------------------------------------------------------
% Standard Ver Test document generator command.
% ------------------------------------------------------------------------------
\newcommand{\stdVerTestDoc}[1]{%
  \stdAutoDocLayout{\stdVerTestTitle{#1}}%
                   {\stdVerTestAbstract{#1}}%
                   {\verTestDir/main.tex}%
}%
