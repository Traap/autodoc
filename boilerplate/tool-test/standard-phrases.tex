% Standard Tool Validation Abstract.
% ------------------------------------------------------------------------------
\newcommand{\stdToolVal}{Tool Validation}
\newcommand{\stdToolValTitle}[1]{#1 \stdToolVal}

\newcommand{\stdToolValAbstract}[1]{%
  This document records validation of #1 as the \stdCfgItem\ identified in this
  \stdRptPkg.  It describes the intended use requirements, execution evidence,
  deviations, and review conclusion used to determine whether the tool is fit for
  its intended use in the controlled software lifecycle.  After the \stdRptPkg\
  has been executed and the evidence supports the conclusion that the applicable
  \stdIUR\ are satisfied, #1 is retained in the \Gls{company} \stdQMS.
}%

% ------------------------------------------------------------------------------
% Standard Tool Validation document generator command.
% ------------------------------------------------------------------------------
\newcommand{\stdToolTestDoc}[1]{%
  \stdAutoDocLayout{\stdToolValTitle{#1}}%
                   {\stdToolValAbstract{#1}}%
                   {\toolTestDir/main.tex}%
}%
