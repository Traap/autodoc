% ------------------------------------------------------------------------------
% Standard Test Report Abstract.
% ------------------------------------------------------------------------------
\newcommand{\stdTestReport}{}
\newcommand{\stdTestReportTitle}[1]{#1 \stdTestReport}

\newcommand{\stdTestReportAbstract}[1]{%
  This document summarizes the executed test results for #1.  It records the
  tested configuration, referenced test plan or protocol, objective evidence,
  deviations, and pass/fail outcome needed to support verification or validation
  conclusions.  After verification, #1 is retained in the \Gls{company} \stdQMS.
}%

% ------------------------------------------------------------------------------
% Standard Test Report document generator command.
% ------------------------------------------------------------------------------
\newcommand{\stdTestReportDoc}[1]{%
  \stdAutoDocLayout{\stdTestReportTitle{#1}}%
                   {\stdTestReportAbstract{#1}}%
                   {\testReportDir/main.tex}%
}%
