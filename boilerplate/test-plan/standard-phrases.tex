% ------------------------------------------------------------------------------
% Standard Test Plan Abstract.
% ------------------------------------------------------------------------------
\newcommand{\stdTestPlan}{test plan}
\newcommand{\stdTestPlanTitle}[1]{#1}

\newcommand{\stdTestPlanAbstract}[1]{%
  This document defines the #1 \stdTestPlan\ for the identified software item or
  test activity.  It describes the test objective, scope, configuration,
  responsibilities, methods, acceptance criteria, and evidence expected from
  execution.  After approval, the plan is retained in \glsentrylong{company}'s
  \glsentrylong{dms} as the controlled basis for test execution and review.
}%

% ------------------------------------------------------------------------------
% Standard Test Plan document generator command. 
% ------------------------------------------------------------------------------
\newcommand{\stdTestPlanDoc}[1]{%
  \stdAutoDocLayout{\stdTestPlanTitle{#1}}%
                   {\stdTestPlanAbstract{#1}}%
                   {\testPlanDir/main.tex}%
}%
